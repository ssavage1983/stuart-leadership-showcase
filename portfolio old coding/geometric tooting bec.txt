  <script>
      document.addEventListener('DOMContentLoaded', () => {
          const lightbox = document.getElementById('tbc-lightbox');
          const lightboxImg = document.getElementById('tbc-lightbox-img');
          const lightboxCaption = document.getElementById('tbc-lightbox-caption');
          const closeBtn = document.querySelector('.tbc-lightbox-close');
          const galleryItems = document.querySelectorAll('.tbc-gallery-item');

          galleryItems.forEach(item => {
              item.addEventListener('click', () => {
                  lightbox.classList.add('active');
                  const img = item.querySelector('.tbc-gallery-img');
                  const caption = item.querySelector('.tbc-gallery-caption');
                  
                  lightboxImg.src = img.src.replace('/v1/fit/w_800', '');
                  lightboxCaption.innerHTML = caption.innerHTML;
              });
          });

          const closeLightbox = () => {
              lightbox.classList.remove('active');
          };

          closeBtn.addEventListener('click', closeLightbox);
          
          lightbox.addEventListener('click', (e) => {
              if (e.target === lightbox) {
                  closeLightbox();
              }
          });

          document.addEventListener('keydown', (e) => {
              if (e.key === 'Escape' && lightbox.classList.contains('active')) {
                  closeLightbox();
              }
          });
      });
  </script>
