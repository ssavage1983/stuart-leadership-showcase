  <!-- LIGHTBOX JAVASCRIPT -->
  <script>
      document.addEventListener('DOMContentLoaded', () => {
          const lightbox = document.getElementById('bgw-lightbox');
          const lightboxContainer = document.getElementById('bgw-lightbox-container');
          const lightboxCaption = document.getElementById('bgw-lightbox-caption');
          const closeBtn = document.querySelector('.bgw-lightbox-close');
          const galleryItems = document.querySelectorAll('.bgw-gallery-item');

          galleryItems.forEach(item => {
              item.addEventListener('click', () => {
                  lightbox.classList.add('active');
                  const media = item.querySelector('.bgw-gallery-media');
                  const caption = item.querySelector('.bgw-gallery-caption');
                  
                  lightboxCaption.innerHTML = caption.innerHTML;
                  lightboxContainer.innerHTML = ''; // Clear previous

                  if (media.tagName.toLowerCase() === 'IMG') {
                      const img = document.createElement('img');
                      img.src = media.src.replace('/v1/fit/w_800', '');
                      img.className = 'bgw-lightbox-element';
                      img.alt = media.alt;
                      lightboxContainer.appendChild(img);
                  } else if (media.tagName.toLowerCase() === 'VIDEO') {
                      const vid = document.createElement('video');
                      vid.src = media.querySelector('source').src;
                      vid.className = 'bgw-lightbox-element';
                      vid.autoplay = true;
                      vid.controls = true;
                      lightboxContainer.appendChild(vid);
                  }
              });
          });

          const closeLightbox = () => {
              lightbox.classList.remove('active');
              lightboxContainer.innerHTML = ''; // Stop video playback
          };

          closeBtn.addEventListener('click', closeLightbox);
          
          lightbox.addEventListener('click', (e) => {
              if (e.target === lightbox || e.target.classList.contains('bgw-lightbox-container')) {
                  closeLightbox();
              }
          });

          document.addEventListener('keydown', (e) => {
              if (e.key === 'Escape' && lightbox.classList.contains('active')) {
                  closeLightbox();
              }
          });
      });
  </script>
