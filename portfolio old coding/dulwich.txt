  <script>
      document.addEventListener('DOMContentLoaded', () => {
          const lightbox = document.getElementById('ds-lightbox');
          const lightboxImg = document.getElementById('ds-lightbox-img');
          const lightboxCaption = document.getElementById('ds-lightbox-caption');
          const closeBtn = document.querySelector('.ds-lightbox-close');
          const galleryItems = document.querySelectorAll('.ds-gallery-item');

          galleryItems.forEach(item => {
              item.addEventListener('click', () => {
                  const img = item.querySelector('.ds-gallery-img');
                  const caption = item.querySelector('.ds-gallery-caption');
                  
                  lightbox.classList.add('active');
                  lightboxImg.src = img.src.replace('/v1/fit/w_800', '');
                  lightboxCaption.innerHTML = caption.innerHTML;
              });
          });

          const closeLightbox = () => {
              lightbox.classList.remove('active');
          };

          closeBtn.addEventListener('click', closeLightbox);
          
          lightbox.addEventListener('click', (e) => {
              if (e.target !== lightboxImg && e.target !== lightboxCaption) {
                  closeLightbox();
              }
          });

          document.addEventListener('keydown', (e) => {
              if (e.key === 'Escape' && lightbox.classList.contains('active')) {
                  closeLightbox();
              }
          });
      });
  </script>
