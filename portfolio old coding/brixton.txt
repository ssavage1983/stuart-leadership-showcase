  <script>
      document.addEventListener('DOMContentLoaded', () => {
          const lightbox = document.getElementById('bj-lightbox');
          const lightboxImg = document.getElementById('bj-lightbox-img');
          const lightboxCaption = document.getElementById('bj-lightbox-caption');
          const closeBtn = document.querySelector('.bj-lightbox-close');
          const galleryItems = document.querySelectorAll('.bj-gallery-item');

          galleryItems.forEach(item => {
              item.addEventListener('click', () => {
                  const img = item.querySelector('.bj-gallery-img');
                  const caption = item.querySelector('.bj-gallery-caption');
                  
                  lightbox.classList.add('active');
                  // Forcing highest quality image string on click
                  lightboxImg.src = img.src.replace('/v1/fit/w_800', '');
                  lightboxCaption.innerHTML = caption.innerHTML;
              });
          });

          const closeLightbox = () => {
              lightbox.classList.remove('active');
          };

          closeBtn.addEventListener('click', closeLightbox);
          
          lightbox.addEventListener('click', (e) => {
              if (e.target !== lightboxImg && e.target !== lightboxCaption) {
                  closeLightbox();
              }
          });

          document.addEventListener('keydown', (e) => {
              if (e.key === 'Escape' && lightbox.classList.contains('active')) {
                  closeLightbox();
              }
          });
      });
  </script>
