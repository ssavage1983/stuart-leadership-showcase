  <script>
      document.addEventListener('DOMContentLoaded', () => {
          const lightbox = document.getElementById('ws-lightbox');
          const lightboxImg = document.getElementById('ws-lightbox-img');
          const lightboxCaption = document.getElementById('ws-lightbox-caption');
          const closeBtn = document.querySelector('.ws-lightbox-close');
          const galleryItems = document.querySelectorAll('.ws-gallery-item');

          galleryItems.forEach(item => {
              item.addEventListener('click', () => {
                  lightbox.classList.add('active');
                  const img = item.querySelector('.ws-gallery-img');
                  const caption = item.querySelector('.ws-gallery-caption');
                  
                  lightboxImg.src = img.src.replace('/v1/fit/w_800', '');
                  lightboxCaption.innerHTML = caption.innerHTML;
              });
          });

          const closeLightbox = () => {
              lightbox.classList.remove('active');
          };

          closeBtn.addEventListener('click', closeLightbox);
          
          lightbox.addEventListener('click', (e) => {
              if (e.target === lightbox) {
                  closeLightbox();
              }
          });

          document.addEventListener('keydown', (e) => {
              if (e.key === 'Escape' && lightbox.classList.contains('active')) {
                  closeLightbox();
              }
          });
      });
  </script>
