  <!-- LIGHTBOX JAVASCRIPT -->
  <script>
      document.addEventListener('DOMContentLoaded', () => {
          const lightbox = document.getElementById('tzg-lightbox');
          const lightboxContainer = document.getElementById('tzg-lightbox-container');
          const lightboxCaption = document.getElementById('tzg-lightbox-caption');
          const closeBtn = document.querySelector('.tzg-lightbox-close');
          const galleryItems = document.querySelectorAll('.tzg-gallery-item');

          galleryItems.forEach(item => {
              item.addEventListener('click', () => {
                  lightbox.classList.add('active');
                  const media = item.querySelector('.tzg-gallery-media');
                  const caption = item.querySelector('.tzg-gallery-caption');
                  
                  lightboxCaption.innerHTML = caption.innerHTML;
                  lightboxContainer.innerHTML = ''; // Clear previous

                  if (media.tagName.toLowerCase() === 'IMG') {
                      const img = document.createElement('img');
                      img.src = media.src.replace('/v1/fit/w_800', '');
                      img.className = 'tzg-lightbox-element';
                      img.alt = media.alt;
                      lightboxContainer.appendChild(img);
                  }
              });
          });

          const closeLightbox = () => {
              lightbox.classList.remove('active');
              lightboxContainer.innerHTML = '';
          };

          closeBtn.addEventListener('click', closeLightbox);
          
          lightbox.addEventListener('click', (e) => {
              if (e.target === lightbox || e.target.classList.contains('tzg-lightbox-container')) {
                  closeLightbox();
              }
          });

          document.addEventListener('keydown', (e) => {
              if (e.key === 'Escape' && lightbox.classList.contains('active')) {
                  closeLightbox();
              }
          });
      });
  </script>
