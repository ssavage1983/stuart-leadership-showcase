  <!-- LIGHTBOX JAVASCRIPT -->
  <script>
      document.addEventListener('DOMContentLoaded', () => {
          const lightbox = document.getElementById('mm-lightbox');
          const lightboxImg = document.getElementById('mm-lightbox-img');
          const lightboxCaption = document.getElementById('mm-lightbox-caption');
          const closeBtn = document.querySelector('.mm-lightbox-close');
          const galleryItems = document.querySelectorAll('.mm-gallery-item');

          galleryItems.forEach(item => {
              item.addEventListener('click', () => {
                  const img = item.querySelector('.mm-gallery-img');
                  const caption = item.querySelector('.mm-gallery-caption');
                  
                  lightbox.classList.add('active');
                  // Forcing highest quality image string on click
                  lightboxImg.src = img.src.replace('/v1/fit/w_800', '');
                  lightboxCaption.innerHTML = caption.innerHTML;
              });
          });

          const closeLightbox = () => {
              lightbox.classList.remove('active');
          };

          closeBtn.addEventListener('click', closeLightbox);
          
          lightbox.addEventListener('click', (e) => {
              if (e.target !== lightboxImg && e.target !== lightboxCaption) {
                  closeLightbox();
              }
          });

          document.addEventListener('keydown', (e) => {
              if (e.key === 'Escape' && lightbox.classList.contains('active')) {
                  closeLightbox();
              }
          });
      });
  </script>
